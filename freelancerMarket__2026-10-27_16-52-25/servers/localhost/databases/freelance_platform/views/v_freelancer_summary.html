DEFINER = 'root'
VIEW v_freelancer_summary
AS
SELECT
  `f`.`id` AS `freelancer_id`,
  CONCAT(`u`.`first_name`, ' ', `u`.`last_name`) AS `full_name`,
  `u`.`email` AS `email`,
  `f`.`location` AS `location`,
  `f`.`registration_date` AS `registration_date`
FROM (`freelancer` `f`
  JOIN `user_account` `u`
    ON (`f`.`user_account_id` = `u`.`id`));
