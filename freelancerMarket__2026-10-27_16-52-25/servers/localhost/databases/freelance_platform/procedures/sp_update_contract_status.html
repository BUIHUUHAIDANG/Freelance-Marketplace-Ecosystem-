CREATE
DEFINER = 'root'
PROCEDURE sp_update_contract_status (IN p_contract_id bigint,
IN p_status_id bigint)
BEGIN
  UPDATE contract
  SET current_contract_status_id = p_status_id
  WHERE id = p_contract_id;
END
