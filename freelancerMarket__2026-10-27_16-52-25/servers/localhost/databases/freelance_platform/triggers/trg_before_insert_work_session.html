CREATE
DEFINER = 'root'
TRIGGER trg_before_insert_work_session
BEFORE INSERT
ON work_session
FOR EACH ROW
BEGIN
  IF NEW.end_time IS NOT NULL
    AND NEW.end_time < NEW.start_time THEN
    SET NEW.end_time = NEW.start_time;
  END IF;
END
